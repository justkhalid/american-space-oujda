    66→export function AdminDashboard({ initialTab = "overview" }: { initialTab?: AdminTab }) {
    67→  const { data: session, status } = useSession();
    68→  const navigate = useRouter((s) => s.navigate);
    69→  const [tab, setTab] = React.useState<AdminTab>(initialTab);
    70→
    71→  React.useEffect(() => setTab(initialTab), [initialTab]);
    72→
    73→  // Auth guard
    74→  React.useEffect(() => {
    75→    if (status === "loading") return;
    76→    if (!session) {
    77→      navigate({ name: "login" });
    78→      return;
    79→    }
    80→    const role = (session.user as { role?: string })?.role;
    81→    if (role === "TEACHER") {
    82→      navigate({ name: "teacher" });
    83→    } else if (role === "EDITOR") {
    84→      navigate({ name: "editor" });
    85→    }
    86→  }, [session, status, navigate]);
    87→
    88→  if (status === "loading" || !session) {
    89→    return (
    90→      <div className="flex items-center justify-center py-32">
    91→        <Loader2 className="w-6 h-6 animate-spin text-muted-foreground" />
    92→      </div>
    93→    );
    94→  }
    95→
    96→  const onTabChange = (t: string) => navigate({ name: "admin-tab", tab: t as AdminTab });
    97→
    98→  return (
    99→    <DashboardLayout
   100→      title="Admin Dashboard"
   101→      pillLabel="Admin"
   102→      pillIcon={ShieldCheck}
   103→      tabs={TABS}
   104→      activeTab={tab}
   105→      onTabChange={onTabChange}
   106→      baseRoute={{ name: "admin" }}
   107→    >
   108→      {tab === "overview" && <OverviewTab />}
   109→      {tab === "applications" && <ApplicationsTab />}
   110→      {tab === "events" && <EventsTab />}
   111→      {tab === "gallery" && <GalleryTab />}
   112→      {tab === "courses" && <CoursesTab />}
   113→      {tab === "clubs" && <ClubsTab />}
   114→      {tab === "links" && <LinksTab />}
   115→      {tab === "members" && <MembersTab />}
   116→      {tab === "registrations" && <RegistrationsTab />}
   117→      {tab === "comments" && <CommentsTab />}
   118→      {tab === "settings" && <SettingsTab />}
   119→      {tab === "users" && <UsersTab />}
   120→    </DashboardLayout>
   121→  );
   122→}

   191→type Status = "PENDING" | "REVIEWING" | "ACCEPTED" | "REJECTED";
   192→interface Application {
   193→  id: string;
   194→  role: string;
   195→  fullName: string;
   196→  email: string;
   197→  phone: string | null;
   198→  city: string | null;
   199→  country: string | null;
   200→  age: number | null;
   201→  occupation: string | null;
   202→  organization: string | null;
   203→  languages: string | null;
   204→  motivation: string;
   205→  experience: string | null;
   206→  status: Status;
   207→  createdAt: string;
   208→}
   209→
   210→const STATUS_COLORS: Record<Status, string> = {
   211→  PENDING: "bg-amber-500/15 text-amber-700 dark:text-amber-300",
   212→  REVIEWING: "bg-sky-500/15 text-sky-700 dark:text-sky-300",
   213→  ACCEPTED: "bg-emerald-500/15 text-emerald-700 dark:text-emerald-300",
   214→  REJECTED: "bg-rose-500/15 text-rose-700 dark:text-rose-300",
   215→};
   216→
   217→function ApplicationsTab() {
   218→  const [apps, setApps] = React.useState<Application[]>([]);
   219→  const [loading, setLoading] = React.useState(true);
   220→  const [filter, setFilter] = React.useState<"ALL" | string>("ALL");
   221→  const [statusFilter, setStatusFilter] = React.useState<"ALL" | Status>("ALL");
   222→
   223→  const load = React.useCallback(() => {
   224→    setLoading(true);
   225→    fetch("/api/applications")
   226→      .then((r) => r.json())
   227→      .then((d) => setApps(d.applications || []))
   228→      .finally(() => setLoading(false));
   229→  }, []);
   230→  React.useEffect(() => load(), [load]);
   231→
   232→  const updateStatus = async (id: string, status: Status) => {
   233→    setApps((a) => a.map((x) => (x.id === id ? { ...x, status } : x)));
   234→    await fetch(`/api/admin/applications?id=${id}`, {
   235→      method: "PATCH",
   236→      headers: { "Content-Type": "application/json" },
   237→      body: JSON.stringify({ status }),
   238→    });
   239→    toast.success(`Marked as ${status.toLowerCase()}.`);
   240→  };
   241→
   242→  const del = async (id: string) => {
   243→    setApps((a) => a.filter((x) => x.id !== id));
   244→    await fetch(`/api/admin/applications?id=${id}`, { method: "DELETE" });
   245→    toast.success("Application deleted.");
   246→  };
   247→
   248→  const filtered = apps
   249→    .filter((a) => filter === "ALL" || a.role === filter)
   250→    .filter((a) => statusFilter === "ALL" || a.status === statusFilter);
   251→
   252→  return (
   253→    <div className="space-y-4">
   254→      <div className="flex flex-wrap items-center gap-3">
   255→        <div className="flex items-center gap-2 flex-wrap">
   256→          <span className="text-xs uppercase tracking-wider text-muted-foreground">Role:</span>
   257→          {["ALL", "TEACHER", "VOLUNTEER", "INTERN", "TRAINER"].map((r) => (
   258→            <button
   259→              key={r}
   260→              onClick={() => setFilter(r)}
   261→              className={`tap px-2.5 py-1 rounded-full text-xs font-medium ${
   262→                filter === r ? "bg-primary text-primary-foreground" : "bg-secondary hover:bg-secondary/70"
   263→              }`}
   264→            >
   265→              {r === "ALL" ? "All" : r.charAt(0) + r.slice(1).toLowerCase()}
   266→            </button>
   267→          ))}
   268→        </div>
   269→        <div className="flex items-center gap-2 ml-auto">
   270→          <span className="text-xs uppercase tracking-wider text-muted-foreground">Status:</span>
   271→          <Select value={statusFilter} onValueChange={(v) => setStatusFilter(v as never)}>
   272→            <SelectTrigger className="w-36 h-8 text-xs">
   273→              <SelectValue />
   274→            </SelectTrigger>
   275→            <SelectContent>
   276→              <SelectItem value="ALL">All</SelectItem>
   277→              <SelectItem value="PENDING">Pending</SelectItem>
   278→              <SelectItem value="REVIEWING">Reviewing</SelectItem>
   279→              <SelectItem value="ACCEPTED">Accepted</SelectItem>
   280→              <SelectItem value="REJECTED">Rejected</SelectItem>
   281→            </SelectContent>
   282→          </Select>
   283→        </div>
   284→      </div>
   285→
   286→      {loading ? (
   287→        <div className="text-center py-16">
   288→          <Loader2 className="w-6 h-6 animate-spin mx-auto text-muted-foreground" />
   289→        </div>
   290→      ) : filtered.length === 0 ? (
   291→        <MatteCard className="text-center py-12">
   292→          <FileText className="w-10 h-10 text-muted-foreground/40 mx-auto mb-3" />
   293→          <p className="text-muted-foreground">No applications match these filters.</p>
   294→        </MatteCard>
   295→      ) : (
   296→        <div className="space-y-3">
   297→          {filtered.map((app) => (
   298→            <AppRow
   299→              key={app.id}
   300→              app={app}
   301→              onStatus={(s) => updateStatus(app.id, s)}
   302→              onDelete={() => del(app.id)}
   303→            />
   304→          ))}
   305→        </div>
   306→      )}
   307→    </div>
   308→  );
   309→}

   354→            </div>
   355→            <div>
   356→              <div className="text-xs uppercase tracking-wider text-muted-foreground mb-1">Background</div>
   357→              <div className="space-y-1">
   358→                {app.age && <div>Age: {app.age}</div>}
   359→                {app.occupation && <div>Occupation: {app.occupation}</div>}
   360→                {app.organization && <div>Organization: {app.organization}</div>}
   361→                {app.languages && <div>Languages: {app.languages}</div>}
   362→              </div>
   363→            </div>
   364→          </div>
   365→          <div>
   366→            <div className="text-xs uppercase tracking-wider text-muted-foreground mb-1">Motivation</div>
   367→            <p className="text-sm leading-relaxed pretty">{app.motivation}</p>
   368→          </div>
   369→          {app.experience && (
   370→            <div>
   371→              <div className="text-xs uppercase tracking-wider text-muted-foreground mb-1">Experience</div>
   372→              <p className="text-sm leading-relaxed pretty">{app.experience}</p>

   376→            <span className="text-xs uppercase tracking-wider text-muted-foreground mr-2">Update:</span>

   378→              <button
   379→                key={s}
   380→                onClick={() => onStatus(s)}
   381→                className={`tap px-2.5 py-1 rounded-full text-xs font-medium ${app.status === s ? STATUS_COLORS[s] : "bg-secondary hover:bg-secondary/70"}`}
   382→              >
   383→                {s.charAt(0) + s.slice(1).toLowerCase()}
   384→              </button>
   385→            ))}
   386→            <button onClick={onDelete} className="tap ml-auto px-2.5 py-1 rounded-full text-xs font-medium text-rose-600 hover:bg-rose-500/10 flex items-center gap-1">
   387→              <Trash2 className="w-3 h-3" /> Delete
   388→            </button>
   389→          </div>
   390→        </div>

   398→// ============================================================
   399→interface EventItem {
   400→  id: string;
   401→  title: string;
   402→  description: string;
   403→  category: string;
   404→  startDate: string;
   405→  endDate: string | null;
   406→  location: string | null;
   407→  capacity: number | null;
   408→  registered: number;
   409→  featured: boolean;
   410→}
   411→
   412→export function EventsTab() {
   413→  const [events, setEvents] = React.useState<EventItem[]>([]);
   414→  const [loading, setLoading] = React.useState(true);
   415→  const [editing, setEditing] = React.useState<EventItem | null>(null);
   416→  const [creating, setCreating] = React.useState(false);
   417→
   418→  const load = React.useCallback(() => {
   419→    setLoading(true);
   420→    fetch("/api/events?upcoming=0&limit=200")
   421→      .then((r) => r.json())
   422→      .then((d) => setEvents(d.events || []))
   423→      .finally(() => setLoading(false));
   424→  }, []);
   425→  React.useEffect(() => load(), [load]);
   426→
   427→  const del = async (id: string) => {
   428→    setEvents((e) => e.filter((x) => x.id !== id));
   429→    await fetch(`/api/events?id=${id}`, { method: "DELETE" });
   430→    toast.success("Event deleted.");
   431→  };
   432→
   433→  if (creating || editing) {
   434→    return (
   435→      <EventEditor
   436→        event={editing}
   437→        onClose={() => {
   438→          setCreating(false);
   439→          setEditing(null);
   440→        }}
   441→        onSaved={() => {
   442→          setCreating(false);
   443→          setEditing(null);
   444→          load();
   445→        }}
   446→      />
   447→    );
   448→  }
   449→
   450→  return (
   451→    <div className="space-y-4">
   452→      <div className="flex items-center justify-between gap-3">
   453→        <div>
   454→          <h2 className="font-display text-xl tracking-tight">Events</h2>
   455→          <p className="text-sm text-muted-foreground">{events.length} events total</p>
   456→        </div>
   457→        <Button onClick={() => setCreating(true)} size="sm" className="rounded-full">
   458→          <Plus className="w-4 h-4" /> New event
   459→        </Button>
   460→      </div>
   461→
   462→      {loading ? (
   463→        <div className="text-center py-16">
   464→          <Loader2 className="w-6 h-6 animate-spin mx-auto text-muted-foreground" />
   465→        </div>
   466→      ) : events.length === 0 ? (
   467→        <MatteCard className="text-center py-12">
   468→          <CalendarDays className="w-10 h-10 text-muted-foreground/40 mx-auto mb-3" />
   469→          <p className="text-muted-foreground">No events yet. Create your first event.</p>
   470→        </MatteCard>
   471→      ) : (
   472→        <div className="space-y-2">
   473→          {events.map((ev) => (
   474→            <MatteCard key={ev.id} className="p-4 flex items-center gap-4">
   475→              <div className="w-12 h-12 rounded-xl bg-primary/8 flex flex-col items-center justify-center shrink-0 text-primary">
   476→                <div className="text-[10px] uppercase font-semibold tracking-wider">{format(new Date(ev.startDate), "MMM")}</div>
   477→                <div className="font-display text-lg tnum leading-tight">{format(new Date(ev.startDate), "d")}</div>
   478→              </div>
   479→              <div className="flex-1 min-w-0">
   480→                <div className="flex items-center gap-2 flex-wrap">
   481→                  <span className="font-medium truncate">{ev.title}</span>
   482→                  <Pill variant="muted">{ev.category}</Pill>
   483→                  {ev.featured && <Pill variant="accent">Featured</Pill>}
   484→                </div>
   485→                <div className="text-xs text-muted-foreground mt-0.5">
   486→                  {format(new Date(ev.startDate), "EEE, MMM d · HH:mm")} · {ev.location || "TBA"} · {ev.registered}/{ev.capacity || "∞"}
   487→                </div>
   488→              </div>
   489→              <div className="flex items-center gap-1">
   490→                <button onClick={() => setEditing(ev)} className="tap w-8 h-8 rounded-lg hover:bg-secondary flex items-center justify-center">
   491→                  <Pencil className="w-3.5 h-3.5" />
   492→                </button>
   493→                <button onClick={() => del(ev.id)} className="tap w-8 h-8 rounded-lg hover:bg-rose-500/10 text-rose-600 flex items-center justify-center">
   494→                  <Trash2 className="w-3.5 h-3.5" />
   495→                </button>
   496→              </div>
   497→            </MatteCard>
   498→          ))}
   499→        </div>
   500→      )}
   501→    </div>
   502→  );
   503→}

   505→function EventEditor({
   506→  event,
   507→  onClose,
   508→  onSaved,
   509→}: {
   510→  event: EventItem | null;
   511→  onClose: () => void;
   512→  onSaved: () => void;
   513→}) {
   514→  const [form, setForm] = React.useState({
   515→    title: event?.title || "",
   516→    description: event?.description || "",
   517→    category: event?.category || "WORKSHOP",

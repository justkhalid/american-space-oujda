   122→   122→export function CompanionDashboard({ initialTab = "overview" }: { initialTab?: CompanionTab }) {
   123→   123→  const { data: session, status } = useSession();
   124→   124→  const navigate = useRouter((s) => s.navigate);
   125→   125→  const [tab, setTab] = React.useState<CompanionTab>(initialTab);
   126→   126→
   127→   127→  React.useEffect(() => setTab(initialTab), [initialTab]);
   128→   128→
   129→   129→  // Auth guard — send unauthenticated users to login.
   130→   130→  React.useEffect(() => {
   131→   131→    if (status === "loading") return;
   132→   132→    if (!session) {
   133→   133→      navigate({ name: "login" });
   134→   134→    }
   135→   135→  }, [session, status, navigate]);
   136→   136→
   137→   137→  if (status === "loading" || !session) {
   138→   138→    return (
   139→   139→      <div className="flex items-center justify-center py-32">
   140→   140→        <Loader2 className="w-6 h-6 animate-spin text-muted-foreground" />
   141→   141→      </div>
   142→   142→    );
   143→   143→  }
   144→   144→
   145→   145→  const onTabChange = (t: string) => navigate({ name: "companion-tab", tab: t as CompanionTab });
   146→   146→
   147→   147→  return (
   148→   148→    <DashboardLayout
   149→   149→      title="ELTASO Companion"
   150→   150→      pillLabel="Companion"
   151→   151→      pillIcon={BookOpen}
   152→   152→      tabs={TABS}
   153→   153→      activeTab={tab}
   154→   154→      onTabChange={onTabChange}
   155→   155→      baseRoute={{ name: "companion" }}
   156→   156→    >
   157→   157→      {tab === "overview" && <OverviewTab onNavigate={(t) => setTab(t)} />}
   158→   158→      {tab === "levels" && <LevelsTab />}

   376→   376→function WeeksList({ level, onChanged }: { level: Level; onChanged: () => void }) {
   377→   377→  const [weeks, setWeeks] = React.useState<Week[]>([]);
   378→   378→  const [settings, setSettings] = React.useState<Record<string, string>>({});
   379→   379→  const [loading, setLoading] = React.useState(true);
   380→   380→  const [editing, setEditing] = React.useState<Week | null>(null);
   381→   381→  const [creating, setCreating] = React.useState(false);
   382→   382→
   383→   383→  const load = React.useCallback(() => {
   384→   384→    setLoading(true);
   385→   385→    Promise.all([
   386→   386→      fetch(`/api/companion/weeks?levelId=${level.id}`).then((r) => r.json()),
   387→   387→      fetch("/api/companion/settings").then((r) => r.json()),
   388→   388→    ])
   389→   389→      .then(([w, s]) => {
   390→   390→        setWeeks(w.weeks || []);
   391→   391→        setSettings(s.settings || {});
   392→   392→      })
   393→   393→      .finally(() => setLoading(false));
   394→   394→  }, [level.id]);
   395→   395→  React.useEffect(() => load(), [load]);
   396→   396→
   397→   397→  const copyWhatsApp = async (w: Week) => {
   398→   398→    const tpl = settings.tpl || "";
   399→   399→    if (!tpl) {
   400→   400→      toast.error("No WhatsApp template set in settings.");
   401→   401→      return;
   402→   402→    }
   403→   403→    const s1Start = settings.s1Start ? new Date(settings.s1Start) : null;
   404→   404→    const weekDate =
   405→   405→      s1Start && w.weekNumber
   406→   406→        ? format(addWeeks(s1Start, w.weekNumber - 1), "MMM d, yyyy")
   407→   407→        : "—";
   408→   408→    const links = w.urls.length > 0 ? w.urls.map((u) => `• ${u}`).join("\n") : "—";
   409→   409→    const msg = tpl
   410→   410→      .replace(/\{teacher\}/g, "")
   411→   411→      .replace(/\{level\}/g, level.label)
   412→   412→      .replace(/\{week\}/g, String(w.weekNumber))
   413→   413→      .replace(/\{date\}/g, weekDate)
   414→   414→      .replace(/\{theme\}/g, w.theme || "—")
   415→   415→      .replace(/\{obj\}/g, w.objectives || "—")
   416→   416→      .replace(/\{lang\}/g, w.language || "—")
   417→   417→      .replace(/\{act\}/g, w.activities || "—")
   418→   418→      .replace(/\{hw\}/g, w.homework || "—")
   419→   419→      .replace(/\{links\}/g, links)
   420→   420→      .replace(/\{coordinator\}/g, settings.coordinator || "");
   421→   421→    try {
   422→   422→      await navigator.clipboard.writeText(msg);
   423→   423→      toast.success("WhatsApp plan copied to clipboard.");
   424→   424→    } catch {
   425→   425→      toast.error("Could not copy — clipboard blocked by browser.");
   426→   426→    }
   427→   427→  };
   428→   428→
   429→   429→  if (loading) {
   430→   430→    return (
   431→   431→      <div className="px-5 pb-5 pt-1 border-t border-border text-center py-10">
   432→   432→        <Loader2 className="w-5 h-5 animate-spin mx-auto text-muted-foreground" />
   433→   433→      </div>
   434→   434→    );
   435→   435→  }
   436→   436→
   437→   437→  if (editing || creating) {
   438→   438→    return (
   439→   439→      <div className="px-5 pb-5 pt-4 border-t border-border">
   440→   440→        <WeekEditor
   441→   441→          week={creating ? null : editing}
   442→   442→          levelId={level.id}
   443→   443→          existingNumbers={weeks.map((w) => w.weekNumber)}
   444→   444→          onClose={() => {
   445→   445→            setEditing(null);
   446→   446→            setCreating(false);
   447→   447→          }}
   448→   448→          onSaved={() => {
   449→   449→            setEditing(null);
   450→   450→            setCreating(false);
   451→   451→            load();
   452→   452→            onChanged();
   453→   453→          }}
   454→   454→        />
   455→   455→      </div>
   456→   456→    );
   457→   457→  }
   458→   458→
   459→   459→  return (
   460→   460→    <div className="px-5 pb-5 pt-4 border-t border-border space-y-3">
   461→   461→      <div className="flex items-center justify-between gap-2">
   462→   462→        <h3 className="font-display text-base tracking-tight">
   463→   463→          {weeks.length} week{weeks.length !== 1 ? "s" : ""}
   464→   464→        </h3>
   465→   465→        <Button
   466→   466→          size="sm"
   467→   467→          variant="outline"
   468→   468→          className="rounded-full bg-transparent"
   469→   469→          onClick={() => setCreating(true)}
   470→   470→        >
   471→   471→          <Plus className="w-3.5 h-3.5" /> New week
   472→   472→        </Button>
   473→   473→      </div>
   474→   474→
   475→   475→      {weeks.length === 0 ? (

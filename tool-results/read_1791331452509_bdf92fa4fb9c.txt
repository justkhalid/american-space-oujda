   122→export function CompanionDashboard({ initialTab = "overview" }: { initialTab?: CompanionTab }) {
   123→  const { data: session, status } = useSession();
   124→  const navigate = useRouter((s) => s.navigate);
   125→  const [tab, setTab] = React.useState<CompanionTab>(initialTab);
   126→
   127→  React.useEffect(() => setTab(initialTab), [initialTab]);
   128→
   129→  // Auth guard — send unauthenticated users to login.
   130→  React.useEffect(() => {
   131→    if (status === "loading") return;
   132→    if (!session) {
   133→      navigate({ name: "login" });
   134→    }
   135→  }, [session, status, navigate]);
   136→
   137→  if (status === "loading" || !session) {
   138→    return (
   139→      <div className="flex items-center justify-center py-32">
   140→        <Loader2 className="w-6 h-6 animate-spin text-muted-foreground" />
   141→      </div>
   142→    );
   143→  }
   144→
   145→  const onTabChange = (t: string) => navigate({ name: "companion-tab", tab: t as CompanionTab });
   146→
   147→  return (
   148→    <DashboardLayout
   149→      title="ELTASO Companion"
   150→      pillLabel="Companion"
   151→      pillIcon={BookOpen}
   152→      tabs={TABS}
   153→      activeTab={tab}
   154→      onTabChange={onTabChange}
   155→      baseRoute={{ name: "companion" }}
   156→    >
   157→      {tab === "overview" && <OverviewTab onNavigate={(t) => setTab(t)} />}
   158→      {tab === "levels" && <LevelsTab />}
   159→      {tab === "classes" && <ClassesTab />}
   160→      {tab === "team" && <TeamTab />}
   161→      {tab === "library" && <LibraryTab />}
   162→    </DashboardLayout>
   163→  );
   164→}

   282→function LevelsTab() {
   283→  const [levels, setLevels] = React.useState<Level[]>([]);
   284→  const [loading, setLoading] = React.useState(true);
   285→  const [expandedId, setExpandedId] = React.useState<string | null>(null);

   289→    fetch("/api/companion/levels")
   290→      .then((r) => r.json())
   291→      .then((d) => setLevels(d.levels || []))
   292→      .finally(() => setLoading(false));
   293→  }, []);
   294→  React.useEffect(() => load(), [load]);
   295→
   296→  if (loading) {
   297→    return (
   298→      <div className="text-center py-16">
   299→        <Loader2 className="w-6 h-6 animate-spin mx-auto text-muted-foreground" />
   300→      </div>
   301→    );
   302→  }
   303→
   304→  return (
   305→    <div className="space-y-4">
   306→      <div>
   307→        <h2 className="font-display text-xl tracking-tight">Levels</h2>
   308→        <p className="text-sm text-muted-foreground">
   309→          {levels.length} levels · click a level to expand its weekly curriculum.
   310→        </p>
   311→      </div>
   312→

   314→        <MatteCard className="text-center py-12">
   315→          <Layers className="w-10 h-10 text-muted-foreground/40 mx-auto mb-3" />
   316→          <p className="text-muted-foreground">No levels configured yet.</p>

   320→          {levels.map((lvl) => (

   322→              key={lvl.id}
   323→              level={lvl}
   324→              expanded={expandedId === lvl.id}

   330→      )}
   331→    </div>
   332→  );

   376→function WeeksList({ level, onChanged }: { level: Level; onChanged: () => void }) {
   377→  const [weeks, setWeeks] = React.useState<Week[]>([]);
   378→  const [settings, setSettings] = React.useState<Record<string, string>>({});
   379→  const [loading, setLoading] = React.useState(true);
   380→  const [editing, setEditing] = React.useState<Week | null>(null);
   381→  const [creating, setCreating] = React.useState(false);
   382→
   383→  const load = React.useCallback(() => {
   384→    setLoading(true);
   385→    Promise.all([
   386→      fetch(`/api/companion/weeks?levelId=${level.id}`).then((r) => r.json()),
   387→      fetch("/api/companion/settings").then((r) => r.json()),
   388→    ])
   389→      .then(([w, s]) => {
   390→        setWeeks(w.weeks || []);
   391→        setSettings(s.settings || {});
   392→      })
   393→      .finally(() => setLoading(false));
   394→  }, [level.id]);
   395→  React.useEffect(() => load(), [load]);
   396→
   397→  const copyWhatsApp = async (w: Week) => {
   398→    const tpl = settings.tpl || "";
   399→    if (!tpl) {
   400→      toast.error("No WhatsApp template set in settings.");
   401→      return;
   402→    }
   403→    const s1Start = settings.s1Start ? new Date(settings.s1Start) : null;
   404→    const weekDate =
   405→      s1Start && w.weekNumber
   406→        ? format(addWeeks(s1Start, w.weekNumber - 1), "MMM d, yyyy")
   407→        : "—";
   408→    const links = w.urls.length > 0 ? w.urls.map((u) => `• ${u}`).join("\n") : "—";
   409→    const msg = tpl
   410→      .replace(/\{teacher\}/g, "")
   411→      .replace(/\{level\}/g, level.label)
   412→      .replace(/\{week\}/g, String(w.weekNumber))
   413→      .replace(/\{date\}/g, weekDate)
   414→      .replace(/\{theme\}/g, w.theme || "—")
   415→      .replace(/\{obj\}/g, w.objectives || "—")
   416→      .replace(/\{lang\}/g, w.language || "—")
   417→      .replace(/\{act\}/g, w.activities || "—")
   418→      .replace(/\{hw\}/g, w.homework || "—")
   419→      .replace(/\{links\}/g, links)
   420→      .replace(/\{coordinator\}/g, settings.coordinator || "");
   421→    try {
   422→      await navigator.clipboard.writeText(msg);
   423→      toast.success("WhatsApp plan copied to clipboard.");
   424→    } catch {
   425→      toast.error("Could not copy — clipboard blocked by browser.");
   426→    }
   427→  };
   428→
   429→  if (loading) {
   430→    return (
   431→      <div className="px-5 pb-5 pt-1 border-t border-border text-center py-10">
   432→        <Loader2 className="w-5 h-5 animate-spin mx-auto text-muted-foreground" />
   433→      </div>
   434→    );
   435→  }
   436→
   437→  if (editing || creating) {
   438→    return (
   439→      <div className="px-5 pb-5 pt-4 border-t border-border">
   440→        <WeekEditor
   441→          week={creating ? null : editing}
   442→          levelId={level.id}
   443→          existingNumbers={weeks.map((w) => w.weekNumber)}
   444→          onClose={() => {
   445→            setEditing(null);
   446→            setCreating(false);
   447→          }}
   448→          onSaved={() => {
   449→            setEditing(null);
   450→            setCreating(false);
   451→            load();
   452→            onChanged();
   453→          }}
   454→        />
   455→      </div>
   456→    );
   457→  }
   458→
   459→  return (
   460→    <div className="px-5 pb-5 pt-4 border-t border-border space-y-3">
   461→      <div className="flex items-center justify-between gap-2">
   462→        <h3 className="font-display text-base tracking-tight">
   463→          {weeks.length} week{weeks.length !== 1 ? "s" : ""}
   464→        </h3>
   465→        <Button
   466→          size="sm"
   467→          variant="outline"
   468→          className="rounded-full bg-transparent"
   469→          onClick={() => setCreating(true)}
   470→        >
   471→          <Plus className="w-3.5 h-3.5" /> New week
   472→        </Button>
   473→      </div>
   474→
   475→      {weeks.length === 0 ? (
   476→        <p className="text-sm text-muted-foreground py-6 text-center">
   477→          No weeks yet. Add the first one.
   478→        </p>
   479→      ) : (
   480→        <div className="grid gap-3">
   481→          {weeks.map((w) => (
   482→            <WeekCard
   483→              key={w.id}
   484→              week={w}
   485→              onEdit={() => setEditing(w)}
   486→              onCopy={() => copyWhatsApp(w)}
   487→              onDelete={async () => {
   488→                await fetch(`/api/companion/weeks?id=${w.id}`, { method: "DELETE" });
   489→                toast.success("Week deleted.");
   490→                load();
   491→                onChanged();
   492→              }}
   493→            />
   494→          ))}
   495→        </div>
   496→      )}
   497→    </div>
   498→  );
   499→}

   561→                  href={u}
   562→                  target="_blank"

   564→                  className="text-primary hover:underline truncate flex items-center gap-1"

   566→                  <ExternalLink className="w-3 h-3 shrink-0" />

   568→                </a>
   569→              ))}
   570→            </div>
   571→          </div>
   572→        )}
   573→      </div>
   574→    </div>
